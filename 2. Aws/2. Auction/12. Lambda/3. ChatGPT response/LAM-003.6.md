Add these imports:

Then add a custom error:
```
class InvalidJsonError(ValueError):
    pass
```
Now add this helper:
```
def parse_json_body(
    body: str | None,
    *,
    is_base64_encoded: bool = False,
) -> Any:
    if body is None or body == "":
        return None
    raw_body = body
    if is_base64_encoded:
        try:
            raw_body = base64.b64decode(body).decode("utf-8")
        except (ValueError, UnicodeDecodeError) as exc:
            raise InvalidJsonError("Request body is not valid Base64.") from exc
    try:
        return json.loads(raw_body)
    except json.JSONDecodeError as exc:
        raise InvalidJsonError("Request body contains invalid JSON.") from exc
```

One improvement: `base64.b64decode()` can sometimes tolerate malformed input. Make it stricter by using:
```
raw_body = base64.b64decode(
    body,
    validate=True,
).decode("utf-8")
```
Then catch:
```
import binascii
```
and:
```
except (binascii.Error, UnicodeDecodeError) as exc:
```
So the better version is:
```
import base64
import binascii
import json
from typing import Any
class InvalidJsonError(ValueError):
    pass
def parse_json_body(
    body: str | None,
    *,
    is_base64_encoded: bool = False,
) -> Any:
    if body is None or body == "":
        return None
    raw_body = body
    if is_base64_encoded:
        try:
            raw_body = base64.b64decode(
                body,
                validate=True,
            ).decode("utf-8")
        except (binascii.Error, UnicodeDecodeError) as exc:
            raise InvalidJsonError(
                "Request body is not valid Base64."
            ) from exc
    try:
        return json.loads(raw_body)
    except json.JSONDecodeError as exc:
        raise InvalidJsonError(
            "Request body contains invalid JSON."
        ) from exc
```
