# Thuật toán Floyd
```text
procedure Floyd;
(*
Tìm đường đi ngắn nhất giữa tất cả các cặp đỉnh
Đầu vào: Đồ thị cho bởi ma trận trọng số a[i,j], i,j = 1,2,...,n.
Đầu ra:
    Ma trận đường đi ngắn nhất giữa các cặp đỉnh
        d[i,j], i,j = 1,2,...,n,
    trong đó d[i,j] cho độ dài đường đi ngắn nhất từ i đến j.
    Ma trận ghi nhận đường đi
        p[i,j], i,j = 1,2,...,n,
    trong đó p[i,j] ghi nhận đỉnh đi trước đỉnh j trong đường đi ngắn nhất từ i đến j.
*)
begin
    (* Khởi tạo *)
    for i := 1 to n do
        for j := 1 to n do
        begin
            d[i,j] := a[i,j];
            p[i,j] := i;
        end;
    (* Bước lặp *)
    for k := 1 to n do
        for i := 1 to n do
            for j := 1 to n do
                if d[i,j] > d[i,k] + d[k,j] then
                begin
                    d[i,j] := d[i,k] + d[k,j];
                    p[i,j] := p[k,j];
                end;
end;
```
