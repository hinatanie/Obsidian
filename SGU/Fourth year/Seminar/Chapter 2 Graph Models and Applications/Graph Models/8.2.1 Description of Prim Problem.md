# Mô tả thuật toán Prim
```
procedure Prim;
begin
    Chọn đỉnh tùy ý r ∈ V;
    Khởi tạo cây T = (V(T), E(T)) với
    V(T) = {r} và E(T) = ∅;
    while |V(T)| < n đỉnh do
    begin
        Gọi (u, v) là cạnh nhỏ nhất với
        u ∈ V(T) và v ∈ V(G) - V(T);
        E(T) ← E(T) ∪ {(u, v)};
        V(T) ← V(T) ∪ {v};
    end
end;
```

* **Thời gian tính:** $O(|V|^2)$



## Thuật toán Prim
```text
procedure Prim;
begin
    (* Bước khởi tạo *)
    Chọn s là một đỉnh nào đó của đồ thị;
    V_H := {s};
    T := ∅;
    d[s] := 0;
    near[s] := s;
    for v ∈ V \ V_H do
    begin
        d[v] := c[s,v];
        near[v] := s;
    end;
    (* Bước lặp *)
    Stop := false;
    while not Stop do
    begin
        Tìm u ∈ V \ V_H thỏa mãn:
            d[u] = min { d[v] : u ∈ V \ V_H };
        V_H := V_H ∪ {u};
        T := T ∪ {(u, near[u])};
        if |V_H| = n then
        begin
            H := (V_H, T);  (* cây khung nhỏ nhất của đồ thị *)
            Stop := true;
        end
        else
            for v ∈ V \ V_H do
                if d[v] > c[u,v] then
                begin
                    d[v] := c[u,v];
                    near[v] := u;
                end;
    end;
end;
```
