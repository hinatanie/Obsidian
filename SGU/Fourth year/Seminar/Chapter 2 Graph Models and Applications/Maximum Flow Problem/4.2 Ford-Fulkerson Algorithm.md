# Thuật toán Ford-Fulkerson (Ford-Fulkerson Algorithm)
Khi đã có luồng cực đại, lát cắt hẹp nhất có thể tìm theo thủ tục mô tả trong chứng minh định lý 1. Sơ đồ của thuật toán Ford - Fulkerson có thể mô tả trong thủ tục sau đây:
```text
procedure Max_Flow;
(* Thuật toán Ford - Fulkerson *)
begin
    (* Khởi tạo: Bắt đầu từ luồng với giá trị 0 *)
    for u ∈ V do
        for v ∈ V do
            f[u,v] := 0;
    Stop := false;
    while not Stop do
    begin
        Find_Path;
        if PathFound then
            Inc_Flow
        else
            Stop := true;
    end;
    < Luồng cực đại trong mạng là f[u,v], u,v ∈ V >
    < Lát cắt hẹp nhất là (V_T, V \ V_T) >
end;
````
## Thủ tục tìm đường tăng luồng (Find Augmenting Path Procedure)
```
procedure Find_Path;
(*
Thủ tục gán nhãn tìm đường tăng luồng
p[v], ε[v] là nhãn của đỉnh v;
V_T - danh sách các đỉnh có nhãn nhưng chưa xét;
c[u,v] - khả năng thông qua của cung (u,v), u,v ∈ V;
f[u,v] - luồng trên cung (u,v), u,v ∈ V
*)
begin
    p[s] := s;
    ε[s] := +∞;
    V_T := V \ {s};
    PathFound := true;
    while V_T ≠ ∅ do
    begin
        u ⇐ V_T;   (* Lấy u từ V_T *)
        for v ∈ V_T do
        begin
            if (c[u,v] > 0) and (f[u,v] < c[u,v]) then
            begin
                p[v] := u;
                ε[v] := min {
                    ε[u],
                    c[u,v] - f[u,v]
                };
                V_T := V_T \ {v};
                (* Nạp v vào danh sách đỉnh có nhãn *)
                if v = t then
                    exit;
            end;
            if (c[v,u] > 0) and (f[v,u] > 0) then
            begin
                p[v] := -u;
                ε[v] := min {
                    ε[u],
                    f[v,u]
                };
                V_T := V_T \ {v};
                (* Nạp v vào danh sách đỉnh có nhãn *)
                if v = t then
                    exit;
            end;
        end;
    end;
    PathFound := false;
end;
```
## Thủ tục tăng luồng (Increase Flow Procedure)
```
procedure Inc_Flow;
(* Tăng luồng theo đường tăng *)
begin
    v := p[t];
    u := t;
    tang := ε[t];
    while u ≠ s do
    begin
        if v > 0 then
            f[v,u] := f[v,u] + tang
        else
        begin
            v := -v;
            f[u,v] := f[u,v] - tang;
        end;
        u := v;
        v := p[u];
    end;
end;
```
