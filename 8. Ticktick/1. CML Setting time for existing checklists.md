```
$projectId = "6a33a006e8bec2aa4d154a75"
$startTime = [DateTimeOffset]::Parse("2026-08-04T13:31:00+07:00")

$projectData = ticktick project data $projectId --json | ConvertFrom-Json

$tasks = $projectData.tasks | Sort-Object {
    if ($_.title -match '^\s*(\d+)\.') {
        [int]$matches[1]
    }
    else {
        9999
    }
}

for ($i = 0; $i -lt $tasks.Count; $i++) {
    $task = $tasks[$i]
    $taskTime = $startTime.AddMinutes($i * 5)
    $dueDate = $taskTime.ToString("yyyy-MM-ddTHH:mm:sszzz")

    Write-Host "Setting '$($task.title)' to $dueDate with reminder"

    ticktick task update $task.id `
        --id $task.id `
        --project $projectId `
        --due-date $dueDate `
        --time-zone "Asia/Ho_Chi_Minh" `
        --reminders "TRIGGER:PT0S"
}
```
